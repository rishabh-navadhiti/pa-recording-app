# Python runtime deps for AI Medical Scribe.
#
# As of Phase 5, transcription (ElevenLabs) and prechart attachment extraction
# run in Node — so requests / elevenlabs / python-dotenv are gone. Only audio
# capture (record.py / probe_duration.py) and markdown→docx (md_to_docx.py) are
# still Python.
#
# Version bounds are compatible-release ranges (installable across Win + Mac);
# Phase 6 bundling freezes exact versions in the packaged runtime.

# Windows: WASAPI loopback capture
PyAudioWPatch>=0.2.12,<0.3 ; sys_platform == "win32"

# macOS: BlackHole capture. record_macos() imports sounddevice + soundfile;
# numpy is their transitive dependency (sounddevice hands the callback an ndarray).
# None are installed on Windows — record.py imports them lazily inside record_macos().
sounddevice>=0.4.6,<0.6 ; sys_platform == "darwin"
soundfile>=0.12,<0.14   ; sys_platform == "darwin"
numpy>=1.24,<3          ; sys_platform == "darwin"

# Cross-platform: WAV→MP3 (record.py) + duration probe (probe_duration.py)
pydub==0.25.1

# Markdown → .docx (md_to_docx.py)
python-docx>=1.1,<2
